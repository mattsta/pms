## The Philosophy

> "The most powerful systems are those that can improve themselves."

Self-building plugins represent the highest level of the plugin architecture:

- **Meta-programming**: Code that writes code
- **Self-extension**: The system grows its own capabilities
- **Self-management**: Plugins that monitor and fix themselves
- **Emergent behavior**: Complex capabilities from simple rules

### plugin-factory/main.py

````python
"""Plugin Factory - Self-Building Meta-Plugin.

This is the most advanced pattern: a plugin that creates other plugins.
It demonstrates:
- Code generation
- Template-based creation
- Source code analysis
- Self-extending systems

The key insight: by enabling plugins to create plugins,
the system becomes infinitely extensible.
"""

import json
import re
from datetime import datetime
from pathlib import Path
from textwrap import dedent, indent
from typing import Any

# =============================================================================
# Configuration
# =============================================================================

_config = {
    "output_directory": "~/.pms/plugins",
}

# Plugin templates
TEMPLATES = {
    "basic": {
        "description": "Simple plugin with one tool",
        "capabilities": [],
        "default_tools": ["example"],
    },
    "api-integration": {
        "description": "REST API integration with CRUD operations",
        "capabilities": ["network"],
        "default_tools": ["list", "get", "create", "update", "delete"],
    },
    "file-processor": {
        "description": "File processing plugin",
        "capabilities": ["filesystem_read", "filesystem_write"],
        "default_tools": ["process", "validate", "transform"],
    },
    "automation": {
        "description": "Event-driven automation",
        "capabilities": ["events"],
        "default_tools": ["trigger", "schedule", "status"],
    },
    "self-managing": {
        "description": "Self-monitoring and self-healing plugin",
        "capabilities": ["events"],
        "default_tools": ["health_check", "diagnose", "repair", "stats"],
    },
}


# =============================================================================
# Code Generation
# =============================================================================


def _generate_manifest(
    name: str,
    description: str,
    tools: list[dict],
    capabilities: list[str] = None,
) -> dict:
    """Generate a plugin manifest."""
    manifest = {
        "name": name,
        "version": "1.0.0",
        "description": description,
        "author": f"Generated by plugin-factory at {datetime.now().isoformat()}",
        "type": "extension",
        "entrypoint": "main.py",
        "sandbox": bool(capabilities),
        "tools": [],
    }

    if capabilities:
        manifest["capabilities"] = capabilities

    for tool in tools:
        tool_def = {
            "name": tool["name"],
            "description": tool.get("description", f"The {tool['name']} tool"),
        }

        if "params" in tool or "parameters" in tool:
            params = tool.get("params") or tool.get("parameters", {})
            tool_def["parameters"] = {}
            for param_name, param_info in params.items():
                if isinstance(param_info, str):
                    tool_def["parameters"][param_name] = {
                        "type": param_info,
                        "required": True,
                    }
                else:
                    tool_def["parameters"][param_name] = param_info

        manifest["tools"].append(tool_def)

    return manifest


def _generate_tool_function(
    name: str,
    description: str,
    parameters: dict = None,
    implementation: str = None,
) -> str:
    """Generate Python code for a tool function."""
    params = []
    param_docs = []

    if parameters:
        for param_name, param_info in parameters.items():
            if isinstance(param_info, str):
                param_type = param_info
                required = True
            else:
                param_type = param_info.get("type", "str")
                required = param_info.get("required", True)

            # Python type mapping
            type_map = {
                "string": "str", "int": "int", "integer": "int",
                "float": "float", "bool": "bool", "boolean": "bool",
                "list": "list", "dict": "dict",
            }
            py_type = type_map.get(param_type, "Any")

            if required:
                params.append(f"{param_name}: {py_type}")
            else:
                params.append(f"{param_name}: {py_type} = None")

            param_docs.append(f"        {param_name}: {param_info.get('description', '')}")

    params_str = ", ".join(params) if params else ""
    params_doc = "\n".join(param_docs) if param_docs else "        None"

    if not implementation:
        implementation = f'return {{"status": "ok", "tool": "{name}"}}'

    return f'''
async def {name}({params_str}) -> dict[str, Any]:
    """{description}

    Args:
{params_doc}

    Returns:
        Result dictionary
    """
    {implementation}
'''


def _generate_main_py(
    name: str,
    description: str,
    tools: list[dict],
) -> str:
    """Generate complete main.py."""
    header = f'''"""{name} - Auto-Generated Plugin.

{description}

Generated by plugin-factory at {datetime.now().isoformat()}

Usage:
    pms plugin discover <directory>
    pms plugin start {name}
'''

    for tool in tools:
        header += f"    pms plugin call {name}.{tool['name']}\n"

    header += '"""\n\nfrom typing import Any\n\n'

    # Generate tool functions
    tool_code = ""
    for tool in tools:
        tool_code += _generate_tool_function(
            name=tool["name"],
            description=tool.get("description", ""),
            parameters=tool.get("params") or tool.get("parameters"),
            implementation=tool.get("implementation"),
        )
        tool_code += "\n"

    # Lifecycle hooks
    lifecycle = f'''
# =============================================================================
# Lifecycle
# =============================================================================


async def on_load(config: dict = None):
    """Initialize plugin."""
    print(f"[{name}] Plugin loaded")


async def on_unload():
    """Cleanup plugin."""
    print(f"[{name}] Plugin unloaded")
'''

    return header + tool_code + lifecycle


# =============================================================================
# Tools
# =============================================================================


async def create_plugin(
    name: str,
    description: str,
    tools: list[dict],
) -> dict[str, Any]:
    """Generate a complete plugin from a specification.

    This is the core self-building capability: describe what you need,
    and a working plugin is created.

    Args:
        name: Plugin name (lowercase, hyphens allowed)
        description: What the plugin should do
        tools: List of tool definitions:
            [{"name": "...", "description": "...", "params": {...}}]

    Returns:
        Information about the created plugin

    Example:
        create_plugin(
            name="my-utils",
            description="My utility functions",
            tools=[
                {"name": "format_date", "description": "Format a date",
                 "params": {"date": "string", "format": "string"}}
            ]
        )
    """
    # Validate name
    if not re.match(r"^[a-z][a-z0-9-]*$", name):
        return {
            "success": False,
            "error": "Invalid name. Use lowercase letters, numbers, hyphens.",
        }

    # Create directory
    output = Path(_config["output_directory"]).expanduser()
    plugin_dir = output / name
    plugin_dir.mkdir(parents=True, exist_ok=True)

    # Generate manifest
    manifest = _generate_manifest(name, description, tools)
    (plugin_dir / "plugin.json").write_text(json.dumps(manifest, indent=2))

    # Generate main.py
    main_code = _generate_main_py(name, description, tools)
    (plugin_dir / "main.py").write_text(main_code)

    # Generate README
    readme = f"""# {name}

{description}

## Installation

```bash
pms plugin discover {output}
pms plugin start {name}
````

## Tools

"""
for tool in tools:
readme += f"### {tool['name']}\n\n{tool.get('description', '')}\n\n"

    (plugin_dir / "README.md").write_text(readme)

    return {
        "success": True,
        "plugin_name": name,
        "path": str(plugin_dir),
        "files": ["plugin.json", "main.py", "README.md"],
        "tools": [t["name"] for t in tools],
        "next_steps": [
            f"pms plugin discover {output}",
            f"pms plugin start {name}",
        ],
    }

async def create_from_template(
template: str,
name: str,
) -> dict[str, Any]:
"""Create a plugin from a predefined template.

    Args:
        template: Template name (basic, api-integration, file-processor, etc.)
        name: Plugin name

    Returns:
        Created plugin information
    """
    if template not in TEMPLATES:
        return {
            "success": False,
            "error": f"Unknown template: {template}",
            "available": list(TEMPLATES.keys()),
        }

    template_info = TEMPLATES[template]

    # Generate tools based on template
    tools = []
    for tool_name in template_info["default_tools"]:
        tools.append({
            "name": tool_name,
            "description": f"{tool_name.title()} operation",
            "params": {"input": "string"},
        })

    return await create_plugin(
        name=name,
        description=f"{name} - created from {template} template",
        tools=tools,
    )

async def analyze_and_generate(
source_path: str,
plugin_name: str,
) -> dict[str, Any]:
"""Analyze source code and generate a plugin wrapper.

    Examines Python files, finds async functions, and creates
    a plugin that exposes them as tools.

    Args:
        source_path: Path to Python file or directory
        plugin_name: Name for generated plugin

    Returns:
        Generated plugin information
    """
    source = Path(source_path).expanduser()

    if not source.exists():
        return {"success": False, "error": f"Source not found: {source_path}"}

    tools = []

    # Find Python files
    if source.is_file():
        files = [source]
    else:
        files = list(source.glob("**/*.py"))

    for file_path in files:
        try:
            content = file_path.read_text()

            # Find async function definitions (simple regex approach)
            pattern = r'async def (\w+)\((.*?)\).*?"""(.*?)"""'
            matches = re.findall(pattern, content, re.DOTALL)

            for func_name, params_str, docstring in matches:
                if not func_name.startswith("_"):  # Skip private
                    tools.append({
                        "name": func_name,
                        "description": docstring.split("\n")[0].strip() if docstring else f"Wrapper for {func_name}",
                        "implementation": f"# TODO: Call original {func_name}\nreturn {{}}",
                    })

        except Exception:
            pass

    if not tools:
        return {
            "success": False,
            "error": "No suitable functions found in source",
        }

    return await create_plugin(
        name=plugin_name,
        description=f"Auto-generated from {source_path}",
        tools=tools,
    )

async def list_templates() -> dict[str, Any]:
"""List available plugin templates.

    Returns:
        Available templates with descriptions
    """
    return {
        "templates": {
            name: {
                "description": info["description"],
                "capabilities": info["capabilities"],
                "default_tools": info["default_tools"],
            }
            for name, info in TEMPLATES.items()
        }
    }

# =============================================================================

# Lifecycle

# =============================================================================

async def on_load(config: dict = None):
"""Initialize the plugin factory."""
global \_config
if config:
\_config.update(config)

    output = Path(_config["output_directory"]).expanduser()
    output.mkdir(parents=True, exist_ok=True)

    print(f"[plugin-factory] Ready to create plugins at {output}")
    print(f"[plugin-factory] Templates: {list(TEMPLATES.keys())}")

async def on_unload():
"""Cleanup."""
print("[plugin-factory] Plugin factory unloaded")


### self-manager/main.py

```python
"""Self-Manager - Self-Monitoring and Self-Healing Plugin.

Demonstrates:
- Health monitoring
- Error detection and recovery
- Automatic issue resolution
- Adaptive behavior

This is the "self-managing" pattern where the plugin
watches itself and takes corrective action.
"""

import asyncio
from datetime import datetime, timedelta
from typing import Any

# =============================================================================
# State
# =============================================================================

_state = {
    "healthy": True,
    "last_check": None,
    "errors": [],
    "repairs": [],
    "alerts": {
        "memory_threshold_mb": 100,
        "error_threshold": 5,
        "check_interval_seconds": 60,
    },
    "stats": {
        "checks_performed": 0,
        "errors_caught": 0,
        "repairs_attempted": 0,
        "repairs_succeeded": 0,
    },
}

_checks = []  # History of health checks


# =============================================================================
# Internal Monitoring
# =============================================================================


def _record_error(error: str, context: dict = None):
    """Record an error for analysis."""
    _state["errors"].append({
        "timestamp": datetime.now().isoformat(),
        "error": error,
        "context": context or {},
    })
    _state["stats"]["errors_caught"] += 1

    # Keep only last 100 errors
    if len(_state["errors"]) > 100:
        _state["errors"] = _state["errors"][-100:]


def _check_error_rate() -> tuple[bool, str]:
    """Check if error rate is acceptable."""
    recent = [
        e for e in _state["errors"]
        if datetime.fromisoformat(e["timestamp"]) > datetime.now() - timedelta(minutes=5)
    ]

    threshold = _state["alerts"]["error_threshold"]
    if len(recent) >= threshold:
        return False, f"High error rate: {len(recent)} errors in 5 minutes"
    return True, "Error rate normal"


def _check_memory() -> tuple[bool, str]:
    """Check memory usage (simulated)."""
    # In production, would use psutil
    import random
    memory_mb = random.randint(20, 80)  # Simulated

    threshold = _state["alerts"]["memory_threshold_mb"]
    if memory_mb > threshold:
        return False, f"Memory usage high: {memory_mb}MB > {threshold}MB"
    return True, f"Memory usage normal: {memory_mb}MB"


def _check_responsiveness() -> tuple[bool, str]:
    """Check if system is responsive."""
    # Simulated check
    return True, "System responsive"


# =============================================================================
# Tools
# =============================================================================


async def health_check() -> dict[str, Any]:
    """Perform comprehensive health check.

    Checks:
    - Error rate
    - Memory usage
    - System responsiveness
    - Internal state consistency

    Returns:
        Health status with details
    """
    _state["stats"]["checks_performed"] += 1
    _state["last_check"] = datetime.now().isoformat()

    checks = {
        "error_rate": _check_error_rate(),
        "memory": _check_memory(),
        "responsiveness": _check_responsiveness(),
    }

    results = {}
    all_healthy = True

    for name, (healthy, message) in checks.items():
        results[name] = {"healthy": healthy, "message": message}
        if not healthy:
            all_healthy = False

    # Record check
    check_record = {
        "timestamp": _state["last_check"],
        "healthy": all_healthy,
        "checks": results,
    }
    _checks.append(check_record)
    if len(_checks) > 100:
        _checks.pop(0)

    _state["healthy"] = all_healthy

    return {
        "healthy": all_healthy,
        "timestamp": _state["last_check"],
        "checks": results,
        "recommendation": None if all_healthy else "Consider running diagnose",
    }


async def diagnose() -> dict[str, Any]:
    """Diagnose issues and suggest fixes.

    Analyzes recent errors and health check failures
    to identify root causes.

    Returns:
        Diagnosis with suggested actions
    """
    diagnosis = {
        "timestamp": datetime.now().isoformat(),
        "issues": [],
        "suggestions": [],
    }

    # Analyze error patterns
    if _state["errors"]:
        error_types = {}
        for error in _state["errors"][-20:]:  # Last 20
            error_msg = error["error"]
            error_types[error_msg] = error_types.get(error_msg, 0) + 1

        for error_msg, count in error_types.items():
            if count > 2:
                diagnosis["issues"].append({
                    "type": "recurring_error",
                    "description": error_msg,
                    "occurrences": count,
                })
                diagnosis["suggestions"].append(
                    f"Investigate recurring error: {error_msg}"
                )

    # Analyze health check history
    recent_checks = _checks[-10:]
    unhealthy_count = sum(1 for c in recent_checks if not c["healthy"])

    if unhealthy_count > 3:
        diagnosis["issues"].append({
            "type": "degraded_health",
            "description": f"{unhealthy_count}/10 recent checks unhealthy",
        })
        diagnosis["suggestions"].append("Consider running repair")

    # Memory trend (simulated)
    if not _check_memory()[0]:
        diagnosis["issues"].append({
            "type": "memory_pressure",
            "description": "Memory usage above threshold",
        })
        diagnosis["suggestions"].append("Clear caches or restart plugin")

    if not diagnosis["issues"]:
        diagnosis["status"] = "healthy"
        diagnosis["message"] = "No issues detected"
    else:
        diagnosis["status"] = "issues_found"
        diagnosis["message"] = f"Found {len(diagnosis['issues'])} issue(s)"

    return diagnosis


async def repair() -> dict[str, Any]:
    """Attempt automatic self-repair.

    Actions:
    - Clear error history
    - Reset internal state
    - Reinitialize connections
    - Clear caches

    Returns:
        Repair results
    """
    _state["stats"]["repairs_attempted"] += 1

    actions_taken = []
    success = True

    try:
        # Action 1: Clear old errors
        old_error_count = len(_state["errors"])
        _state["errors"] = _state["errors"][-10:]  # Keep only recent
        actions_taken.append(f"Cleared {old_error_count - len(_state['errors'])} old errors")

        # Action 2: Reset health state
        _state["healthy"] = True
        actions_taken.append("Reset health state")

        # Action 3: Clear check history
        _checks.clear()
        actions_taken.append("Cleared check history")

        # Action 4: In production, would do things like:
        # - Reconnect to databases
        # - Refresh credentials
        # - Clear caches
        actions_taken.append("Refreshed internal state")

        _state["stats"]["repairs_succeeded"] += 1

    except Exception as e:
        success = False
        actions_taken.append(f"Error during repair: {e}")

    # Record repair
    _state["repairs"].append({
        "timestamp": datetime.now().isoformat(),
        "success": success,
        "actions": actions_taken,
    })

    return {
        "success": success,
        "actions_taken": actions_taken,
        "recommendation": "Run health_check to verify" if success else "Manual intervention needed",
    }


async def stats() -> dict[str, Any]:
    """Get internal statistics.

    Returns:
        Plugin statistics and history
    """
    return {
        "healthy": _state["healthy"],
        "last_check": _state["last_check"],
        "statistics": _state["stats"],
        "recent_errors": len(_state["errors"]),
        "recent_repairs": len(_state["repairs"]),
        "alert_thresholds": _state["alerts"],
    }


async def configure_alerts(
    memory_threshold: int = None,
    error_threshold: int = None,
) -> dict[str, Any]:
    """Configure alert thresholds.

    Args:
        memory_threshold: Memory alert threshold in MB
        error_threshold: Error count threshold

    Returns:
        Updated configuration
    """
    if memory_threshold is not None:
        _state["alerts"]["memory_threshold_mb"] = memory_threshold
    if error_threshold is not None:
        _state["alerts"]["error_threshold"] = error_threshold

    return {
        "configured": True,
        "thresholds": _state["alerts"],
    }


# =============================================================================
# Event Handlers
# =============================================================================


async def on_error(error: str, plugin_name: str = None, **kwargs):
    """Handle error events from the system.

    Records errors and triggers self-healing if needed.
    """
    _record_error(error, {"plugin": plugin_name, **kwargs})

    # Check if we should auto-repair
    error_healthy, _ = _check_error_rate()
    if not error_healthy:
        print("[self-manager] High error rate detected - initiating auto-repair")
        await repair()


async def on_heartbeat(**kwargs):
    """Handle periodic heartbeat events.

    Performs routine health checks.
    """
    await health_check()


# =============================================================================
# Lifecycle
# =============================================================================


async def on_load(config: dict = None):
    """Initialize self-manager."""
    if config:
        if "alerts" in config:
            _state["alerts"].update(config["alerts"])

    print("[self-manager] Self-managing plugin initialized")
    print(f"[self-manager] Alert thresholds: {_state['alerts']}")

    # Initial health check
    result = await health_check()
    print(f"[self-manager] Initial health: {'✓ Healthy' if result['healthy'] else '✗ Issues detected'}")


async def on_unload():
    """Cleanup."""
    print(f"[self-manager] Final stats: {_state['stats']}")
```

## The Self-Building Loop

```
     ┌──────────────────────────────────────┐
     │                                      │
     │    User identifies need              │
     │              │                       │
     │              ▼                       │
     │    Plugin Factory generates          │
     │              │                       │
     │              ▼                       │
     │    New capability available          │
     │              │                       │
     │              ▼                       │
```
